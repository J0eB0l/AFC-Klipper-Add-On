# Stock Bambu Lab AMS units through a BridgeBox, as one pooled chain.
#
# The BridgeBox is the master of the Bambu RS-485 bus and drives stock AMS,
# AMS 2 Pro and AMS HT units on it (the AMS Lite is not supported). Klipper
# talks to it over USB or WiFi. Neither the BridgeBox nor the AMS units are
# an [mcu]: its port goes in serial_port below and nowhere else. Use one
# BridgeBox per bus.
#
# [AFC_BridgeBox chain1] below is the only section you need. At startup it
# builds a pool of bays: four-lane AMS bays and one-lane AMS HT bays. Once
# AFC's PREP has run, each unit on the bus claims its saved bay, or a free
# one of its kind, and a unit plugged in later claims one live, with no
# restart. Its lanes, T#s and panel entry appear on their own.
#
# Before the first boot:
#   1. Copy this file to AFC/AFC_BridgeBox.cfg in your Klipper config
#      folder, where AFC's [include AFC/*.cfg] loads it.
#   2. Set serial_port and extruder below. That extruder needs a toolhead
#      sensor (see the extruder note).
#   3. Settle pool_ams, the bay names and lane_base now: changing them
#      later can move lanes and T#s.
# After boot:
#   4. AFC_BAMBU_UIDS lists the units on the bus and the bridge firmware;
#      AFC_BRIDGEBOX_BAYS shows the bay each unit claimed.
#
# The chain builds the [AFC_BambuAMS] unit, [AFC_lane] and [AFC_hub]
# sections for each bay, a [temperature_sensor] card (temperature and
# humidity) for each recorded unit, and one [AFC_buffer] for the chain.
# Write none of them for Bambu units, and keep your other sections off the
# Bambu lane numbers and bay names: a clash can stop Klipper from
# starting.
#
# There is no bowden length to set. Each unit measures its own filament
# path on its first load, and the chain keeps the figure for that unit.
#
# The chain saves what it learns (the units it has seen, their names, lanes
# and measured lengths) at the end of the file that holds this section, as
# lines starting with #~#. Klipper reads them as comments. Leave them alone
# and keep your own lines above them: everything below the managed-state
# marker line is rewritten.
#
# BridgeBoxes ship flashed. To update one, follow
# Firmwares/BridgeBox/BRIDGE_FLASH_HOWTO.md with the signed_s3.bin image
# and its .manifest.json from the same folder. The AMS units' own firmware
# updater is in Firmwares/Bambu_AMS/.
#
# docs/bambu_hotswap.md covers the pool, the lane layout, the popups and the
# AFC_BRIDGEBOX_* commands in full.


[AFC_BridgeBox chain2]
# ---- Connection ----
# Use ONE of the two blocks below and leave the other commented out.
#
# USB (default): the BridgeBox's USB port. Find it with:
#   ls /dev/serial/by-id/
serial_port: /dev/serial/by-id/usb-Espressif_USB_JTAG_serial_debug_unit_XXXXXXXXXXXX-if00

# WiFi (TCP): a BridgeBox on your network. To use it, comment out the USB
# serial_port line above, uncomment the two lines below, and change:
#   BRIDGE_ADDRESS  the BridgeBox's IP address or hostname. Give it a
#                   fixed DHCP lease so it doesn't move.
#   8888            the BridgeBox's port. 8888 is the default, so ":8888"
#                   can be left off.
#   LINK_KEY        the link key stored on the BridgeBox. Delete the
#                   tcp_key line if it has no key.
# USB is the supported transport; TCP is opt-in.
#serial_port: tcp://BRIDGE_ADDRESS:8888
#tcp_key: LINK_KEY

# ---- Extruder ----
# The [AFC_extruder] this chain feeds. One chain feeds one extruder. The
# AMS feeds each load until that extruder's pin_tool_start sensor sees
# filament, so it needs one (or pin_tool_start: buffer, see buffer_type
# below). A second BridgeBox makes a second chain: add another
# [AFC_BridgeBox <name>] below this one, with its own serial_port and an
# extruder. Its units are named Bambu_AMS_<name>_1 and so on by themselves;
# leave ams_names and ht_names out of it, or give it names of its own.
extruder: extruder

# ---- Pool ----
# Bays built at startup, counting units already recorded. pool_ams is the
# number of four-lane AMS bays (at most 4, the most a Bambu bus takes) and
# pool_ht the number of one-lane AMS HT bays (keep it at 8 or below). Set
# pool_ams to the most AMS you will run: raising it later moves every HT
# lane and T# up. Both at 0 turn live claiming off, and then no unit's
# lanes come up by themselves: read "Turning live claiming off" in
# docs/bambu_hotswap.md first.
pool_ams: 4
pool_ht: 8

# ---- Bay names ----
# Lowest lanes first. A name is what UNIT= takes in the AFC_BAMBU_*
# commands and what the panel shows. Unset, the bays are Bambu_AMS_1 to
# Bambu_AMS_4 and Bambu_AMS_HT_1 to Bambu_AMS_HT_8, and past the end of a
# list the default names carry on. Only four ams_names are used. Renaming
# or reordering them later can move a unit to other lanes and T#s.
#ams_names: Bambu_AMS_1, Bambu_AMS_2, Bambu_AMS_3, Bambu_AMS_4
#ht_names: Bambu_AMS_HT_1, Bambu_AMS_HT_2, Bambu_AMS_HT_3, Bambu_AMS_HT_4, Bambu_AMS_HT_5, Bambu_AMS_HT_6, Bambu_AMS_HT_7, Bambu_AMS_HT_8
# Prefix of the default names. Unset, it is Bambu_AMS, or
# Bambu_AMS_<chain name> on a second chain, so the two chains' names differ.
#unit_prefix: Bambu_AMS

# ---- Lanes and T# ----
# Each Bambu lane's T# is its lane number: lane24 is T24. AMS bay N takes
# the four lanes from lane_base + 4 x (N - 1), and the HT bays follow the
# AMS band, one lane each. With a base of 24 and the default pools,
# Bambu_AMS_1 is lane24-lane27 (T24-T27), Bambu_AMS_4 is lane36-lane39
# and Bambu_AMS_HT_1 is lane40.
#
# lane_base 0 (the default) picks the base at the first start: one past the
# highest laneN in your other [AFC_lane] / [AFC_stepper] sections, else one
# past the highest T# (or [AFC_extruder eN]) in use, else 24. Beside an ACE
# on lane0-lane3 that gives 4: Bambu_AMS_1 is lane4-lane7 (T4-T7) and the
# HT lanes start at lane20. The base is then saved, so lanes added
# elsewhere later do not move the Bambu ones; keep those off the Bambu
# lanes. To put the Bambu lanes elsewhere, set lane_base before the first
# start: changing it later moves every Bambu lane and T#, and those lanes
# lose their saved spool details.
#lane_base: 0

# ---- Known units ----
# A comma-separated list of <model>:<uid> pairs. The model is ams1 (AMS),
# ams2 (AMS 2 Pro), ht (AMS HT) or boxed (an AMS of either generation).
# Leave it unset: the chain records each unit it sees. When set, it
# replaces that record, and a unit it does not list still works but gets
# no saved bay. AFC_BAMBU_UIDS lists the UIDs on the bus.
#roster: ht:0123456789ABCDEF00003331

# ---- Unplugging ----
# Drop an unplugged unit's lanes and T#s live once it has been offline for
# release_grace seconds (minimum 2). Never during a print, and never while
# one of its lanes is loaded to the toolhead. With False (the default) they
# stay until the next restart. A unit that stayed online for 15 s has its
# bay saved, so plugged back in it returns to the same lanes and T#s.
auto_drop: True
#release_grace: 10

# ---- State file ----
# Where the #~# lines are kept. Default: the file that holds this section
# (found in your config folder), else
# ~/printer_data/config/AFC/AFC_BridgeBox.cfg.
#state_file:

# ---- Buffer ----
# The chain's buffer, [AFC_buffer <unit_prefix>_Buffer] (Bambu_AMS_Buffer
# by default), reads each AMS's own buffer through the bridge. FPS_PSF
# (the default) follows its tension. bambu also watches the AMS odometer,
# so the buffer can stand in for a toolhead filament sensor: use it with
# pin_tool_start: buffer in the [AFC_extruder] named above.
buffer_type: bambu

# ---- Dryer ----
# Drying ceiling for every heated unit, in C. 0 (the default) uses each
# model's own: AMS 2 Pro 65, AMS HT 85. A higher value never raises it.
#dry_max_temp: 0

# ---- Unit options ----
# An [AFC_BambuAMS] option set here applies to every unit this chain
# builds; the model and bay sections below win over it. The options are
# listed at the top of extras/AFC_BambuAMS.py. Keep unit_uid, ams_index
# and the mc_* options out of here: unit_uid stops Klipper, and the
# others address one unit, not a whole chain.
#
# Create a Spoolman spool for a tag Spoolman does not know yet. Needs the
# [AFC_BambuAMS_rfid] section below.
#auto_spoolman_create: False
#
# On a jam mid-print, unload, reload the same spool and resume the print.
# On by default for BridgeBox units; False leaves the print paused for you.
# auto_error_recovery_limit is how many resumes in a row it may make by
# itself before it leaves the print paused (0: recover the filament, but
# never resume).
#auto_error_recovery: True
#auto_error_recovery_limit: 1


# Settings for one model. Name the section after the model: ams1 (AMS),
# ams2 (AMS 2 Pro) or ht (AMS HT). A bay name, as in
# [AFC_BridgeBox Bambu_AMS_1], targets that bay alone. Most [AFC_BambuAMS]
# options can go here, and they win over the chain section. Never give
# these a serial_port: a section with one is another chain. A new AMS
# learns its model from the bus after it is recorded: measure_on_insert,
# heater and dry_max_temp set here apply to it then, most others from the
# next restart.
#
# measure_on_insert measures how much filament a spool holds when it is
# inserted; the unit pulls filament for 8 to 25 s to do it. It is on by
# default for every model, including a unit claimed live onto a spare bay.
# Set it False here to turn it off for a model. AFC_BAMBU_CAPSCAN
# LANE=<lane> measures on demand either way.
#[AFC_BridgeBox ht]
#measure_on_insert: False

#[AFC_BridgeBox ams2]
#measure_on_insert: False

#[AFC_BridgeBox ams1]
#measure_on_insert: False


# ---- Spoolman ----
# The AMS reads Bambu tags itself, so the units load, scan and name their
# filament without this. The section adds the Spoolman side: matching the
# spool by its tag, and writing the remaining weight the AMS measures back
# to it (sync_measured_to_spoolman: False in the chain section stops
# that). Needs Spoolman set up in moonraker.conf.
#[AFC_BambuAMS_rfid]


# ---- Drying and other commands ----
# Drying (AMS 2 Pro and AMS HT): templates/AFC_Dryer.cfg adds a web panel
# that finds every heated unit by itself. From the console:
#   AFC_BAMBU_HEATER_START UNIT=Bambu_AMS_HT_1 TEMP=65 TIME=480
#   AFC_BAMBU_HEATER_STOP UNIT=Bambu_AMS_HT_1
# TEMP is in C, capped at the unit's ceiling; TIME is in minutes.
#
#   AFC_BRIDGEBOX_ASSIGN  move a unit to another bay
#   AFC_BRIDGEBOX_FORGET  retire a unit and free its bay
